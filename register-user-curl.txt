# Lệnh curl để đăng ký tài khoản Admin

# Cách 1: Đăng ký trực tiếp qua domain (khi đã cấu hình SSL)
curl -X POST "https://traigabaanhem.com/api/auth/register" \
  -H "Content-Type: application/json" \
  -d '{
    "username": "admin",
    "password": "your_secure_password",
    "email": "admin@traigabaanhem.com"
  }'

# Cách 2: Đăng ký cục bộ qua localhost port 8080 (ngay sau khi docker start)
curl -X POST "http://localhost:8080/api/auth/register" \
  -H "Content-Type: application/json" \
  -d '{
    "username": "admin",
    "password": "your_secure_password",
    "email": "admin@traigabaanhem.com"
  }'
